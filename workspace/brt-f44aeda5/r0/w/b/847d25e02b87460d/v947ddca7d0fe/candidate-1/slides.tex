% SOURCE-BEGIN U0004 0 12

% SOURCE-END U0004

{\normalsize
\textbf{证明：}
% SOURCE-BEGIN U0005 0 136

% SOURCE-END U0005

{\normalsize
% SOURCE-BEGIN U0012 0 49
