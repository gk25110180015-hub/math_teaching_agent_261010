\section{数列极限}
\begin{frame}{数列极限的定义}
\begin{block}{例题}\normalsize 证明 $a_n=1/n$ 收敛于零。\end{block}
\pause {\normalsize\textbf{证明：}任取 $\varepsilon>0$，取 $N>1/\varepsilon$。当 $n>N$，有 $|a_n|<\varepsilon$。}
